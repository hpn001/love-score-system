<script>
/* ===== 数据 ===== */
const DB = {
  get: () => JSON.parse(localStorage.getItem('love_rank')||'null'),
  save(d){ localStorage.setItem('love_rank',JSON.stringify(d)) }
};

if(!DB.get()){
  DB.save({
    me:{ name:'管理员(我)', pwd:'admin888', credits:1000, isAdmin:true },
    mate:{ name:'宝宝', pwd:'123456', credits:500, isAdmin:false },
    tasks:[],
    logs:[]
  });
}

let me=DB.get().me, mate=DB.get().mate, tasks=DB.get().tasks, logs=DB.get().logs;
let current=null;

/* ===== 登录 ===== */
function login(name,pwd){
  if(name===me.name&&pwd===me.pwd) return current=me;
  if(name===mate.name&&pwd===mate.pwd) return current=mate;
  return null;
}
function persist(){
  const d=DB.get();
  d.tasks=tasks; d.me=me; d.mate=mate; d.logs=logs;
  DB.save(d);
}
function esc(s){const d=document.createElement('div');d.textContent=s;return d.innerHTML;}

/* ===== 日志 ===== */
function addLog(user,type,amount,reason){
  logs.unshift({
    time:new Date().toLocaleString(),
    user:user.name,
    type,
    amount,
    reason
  });
  if(logs.length>300) logs.pop();
}

/* ===== 图片 ===== */
function readFileAsDataURL(file){
  return new Promise((ok,no)=>{
    if(!file) return ok(null);
    if(file.size>2.5 * 1024 * 1024) return no(new Error('图片建议小于2MB'));
    const r=new FileReader();
    r.onload=()=>ok(r.result);
    r.onerror=()=>no(r.error);
    r.readAsDataURL(file);
  });
}
function showImage(src){
  const box=document.createElement('div');
  box.className='lightbox';
  box.innerHTML=`<img src="${src}">`;
  box.onclick=()=>box.remove();
  document.body.appendChild(box);
}

/* ===== 排行榜 ===== */
function rankBoard(){
  const list=[me,mate].sort((a,b)=>b.credits-a.credits);
  const medals=['🥇','🥈','🥉'];
  return `
  <div class="card">
    <h2>🏆 积分排行榜</h2>
    <div class="rank">
      ${list.map((u,i)=>`
        <div class="rank-item rank-${i+1}">
          <div class="crown">${medals[i]||'🎖️'}</div>
          <div style="font-size:18px">${esc(u.name)}</div>
          <div class="score">${u.credits}</div>
          <div style="font-size:12px;color:#aaa">${i===0?'当前领先':'加油追赶'}</div>
        </div>
      `).join('')}
    </div>
  </div>`;
}

/* ===== 渲染 ===== */
function render(){
  const app=document.getElementById('app');
  if(!current){ app.innerHTML=loginUI(); return; }

  app.innerHTML=`
  <div class="card topbar">
    <div>👤 <b>${current.name}</b>（积分：<b>${current.credits}</b>）</div>
    <button class="btn-danger btn-sm" onclick="current=null;render()">退出</button>
  </div>
  ${rankBoard()}
  ${current.isAdmin?adminPanel():''}
  ${taskTable()}
  ${logTable()}
  `;
}

/* ===== 登录 ===== */
function loginUI(){return `
<div class="card">
  <h2>登录</h2>
  <input id="u" placeholder="用户名">
  <input id="p" type="password" placeholder="密码">
  <button class="btn" style="width:100%" onclick="
    if(!login(u.value.trim(),p.value)) return alert('账号或密码错误');
    render();
  ">登录</button>
  <p style="font-size:12px;color:#aaa">
    管理员：${me.name} / ${me.pwd}<br>
    宝宝：${mate.name} / ${mate.pwd}
  </p>
</div>`;}

/* ===== 管理员 ===== */
function adminPanel(){return `
<div class="card">
  <h2>👑 管理员操作区</h2>
  <div style="display:flex;gap:10px;flex-wrap:wrap;margin-bottom:10px">
    <select id="target">
      <option value="mate">${mate.name}</option>
    </select>
    <input id="adjAmount" type="number" placeholder="积分数量" style="width:120px">
    <input id="adjReason" placeholder="原因（必填）" style="flex:1">
  </div>
  <div style="margin-bottom:12px">
    <button class="btn btn-sm" onclick="manualAdj('add')">➕ 奖励</button>
    <button class="btn btn-danger btn-sm" onclick="manualAdj('deduct')">➖ 扣除</button>
  </div>
  <hr style="border-color:#2a3a5c">
  <div style="margin-top:12px">
    <input id="tTitle" placeholder="任务标题">
    <input id="tReward" type="number" placeholder="任务奖励积分">
    <button class="btn btn-sm" onclick="publish()">发布任务</button>
  </div>
</div>`;}

/* ===== 手动调分 ===== */
function manualAdj(type){
  const uid=document.getElementById('target').value;
  const amount=parseInt(document.getElementById('adjAmount').value)||0;
  const reason=document.getElementById('adjReason').value.trim();
  if(!amount||!reason) return alert('数量和原因都要填');

  const target=uid==='mate'?mate:me;
  if(type==='add'){
    target.credits+=amount;
    addLog(target,'add',amount,reason);
  }else{
    target.credits=Math.max(0,target.credits-amount);
    addLog(target,'deduct',-amount,reason);
  }
  persist(); render();
}

/* ===== 发布任务 ===== */
function publish(){
  const title=document.getElementById('tTitle').value.trim();
  const reward=parseInt(document.getElementById('tReward').value)||0;
  if(!title||!reward) return alert('填完整');
  tasks.push({id:Date.now(),title,reward,submitted:false,approved:null,proof:'',remark:''});
  persist(); render();
}

/* ===== 任务 ===== */
function taskTable(){
  if(!tasks.length) return '<div class="card">暂无任务</div>';

  return `<div class="card"><h2>📋 任务列表</h2>
  <table>
  <tr><th>任务</th><th>奖励</th><th>状态</th><th>证据</th><th>操作</th></tr>
  ${tasks.map((t,i)=>{
    let state='待做';
    if(t.submitted&&t.approved===null) state='<span class="badge b-wait">待审核</span>';
    if(t.approved===true) state='<span class="badge b-yes">已通过</span>';
    if(t.approved===false) state='<span class="badge b-no">已驳回</span>';

    let proof='—';
    if(t.proof){
      proof=`<img src="${t.proof}" class="preview-img" onclick="showImage('${t.proof}')">
             <div style="font-size:11px">${t.remark||''}</div>`;
    }

    let op='';
    if(!current.isAdmin){
      if(!t.submitted||t.approved===false){
        op=`<button class="btn btn-sm" onclick="submitTask(${i})">提交完成</button>`;
      }
    }else{
      if(t.submitted&&t.approved===null){
        op=`
        <button class="btn-success btn-sm" onclick="audit(${i},true)">通过</button>
        <button class="btn-danger btn-sm" onclick="audit(${i},false)">驳回</button>
        `;
      }
    }

    return `<tr>
      <td>${esc(t.title)}</td>
      <td>+${t.reward}</td>
      <td>${state}</td>
      <td>${proof}</td>
      <td>${op}</td>
    </tr>`;
  }).join('')}
  </table></div>`;
}

/* ===== 提交 ===== */
function submitTask(i){
  const wrap=document.createElement('div');
  wrap.style.cssText='position:fixed;inset:0;background:#0008;display:grid;place-items:center;z-index:99';
  wrap.innerHTML=`
  <div style="background:#16213e;padding:20px;border-radius:12px;width:320px">
    <h3>提交完成证明</h3>
    <input id="r" style="width:100%" placeholder="备注">
    <input id="f" type="file" accept="image/*" style="margin-top:8px">
    <div style="margin-top:10px;text-align:right">
      <button class="btn-danger btn-sm" onclick="this.closest('[style*=fixed]').remove()">取消</button>
      <button class="btn btn-sm" id="go">提交</button>
    </div>
  </div>`;
  document.body.appendChild(wrap);

  wrap.querySelector('#go').onclick=async ()=>{
    const file=wrap.querySelector('#f').files[0];
    const remark=wrap.querySelector('#r').value.trim();
    let base64='';
    try{base64=await readFileAsDataURL(file);}catch(e){return alert(e.message);}
    tasks[i].submitted=true;
    tasks[i].approved=null;
    tasks[i].proof=base64;
    tasks[i].remark=remark;
    persist(); wrap.remove(); render();
  };
}

/* ===== 审核 ===== */
function audit(i,ok){
  if(ok){
    tasks[i].approved=true;
    mate.credits+=tasks[i].reward;
    addLog(mate,'task',tasks[i].reward,'完成任务：'+tasks[i].title);
  }else{
    tasks[i].approved=false;
  }
  persist(); render();
}

/* ===== 积分记录 ===== */
function logTable(){
  if(!logs.length) return '';
  return `<div class="card">
  <h2>📜 积分变动记录</h2>
  <table>
  <tr><th>时间</th><th>用户</th><th>变动</th><th>原因</th></tr>
  ${logs.map(l=>{
    let cls='';
    if(l.type==='add') cls='log-add';
    if(l.type==='deduct') cls='log-deduct';
    if(l.type==='task') cls='log-task';
    const sign=l.amount>0?'+'+l.amount:l.amount;
    return `<tr>
      <td>${l.time}</td>
      <td>${esc(l.user)}</td>
      <td class="${cls}">${sign}</td>
      <td>${esc(l.reason)}</td>
    </tr>`;
  }).join('')}
  </table>
  </div>`;
}

render();
</script>
